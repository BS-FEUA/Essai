    49→// ─── Relative time helper ─────────────────────────
    50→function relativeTime(dateStr: string): string {
    51→  const now = Date.now();
    52→  const then = new Date(dateStr).getTime();
    53→  const diffMs = now - then;
    54→  const diffMin = Math.floor(diffMs / 60000);
    55→  const diffH = Math.floor(diffMs / 3600000);
    56→  const diffD = Math.floor(diffMs / 86400000);
    57→  if (diffMin < 1) return "À l'instant";
    58→  if (diffMin < 60) return `Il y a ${diffMin}min`;
    59→  if (diffH < 24) return `Il y a ${diffH}h`;
    60→  if (diffD < 7) return `Il y a ${diffD}j`;
    61→  return new Date(dateStr).toLocaleDateString('fr-FR');
    62→}

   120→    if (announcements.length <= 1) return;
   121→    const t = setInterval(() => setIdx((i) => (i + 1) % announcements.length), 5000);
