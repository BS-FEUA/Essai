    99→// ─── Announcements banner ───────────────────────────
   100→function AnnouncementsBanner({ announcements }: { announcements: any[] }) {
   101→  const [idx, setIdx] = useState(0);
   102→  useEffect(() => {
   103→    if (announcements.length <= 1) return;
   104→    const t = setInterval(() => setIdx((i) => (i + 1) % announcements.length), 5000);
   105→    return () => clearInterval(t);
   106→  }, [announcements.length]);
   107→  if (announcements.length === 0) return null;
   108→  const a = announcements[idx];
   109→  return (
   110→    <div className={`py-2 px-4 text-center text-sm font-medium ${a?.isUrgent ? 'bg-red-500/10 text-red-700 border-b border-red-200' : 'bg-[var(--feua-green-light)] text-[var(--feua-green)] border-b border-green-200'}`}>
   111→      <AnimatePresence mode="wait">
   112→        <motion.p key={a?.id || idx} {...pageVariants} className="truncate">
   113→          {a?.isUrgent && <AlertTriangle className="inline h-4 w-4 mr-1" />}
   114→          {a?.title}{a?.content ? ` — ${a.content}` : ''}
   115→        </motion.p>
   116→      </AnimatePresence>
   117→    </div>
   118→  );
   119→}
